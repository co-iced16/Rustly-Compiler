// 测试行注释
fn compute() -> i32 {
    // 这是一个单行注释
    let x: i32 = 10; // 行尾注释
    let y: i32 = 20;
    // 中间的注释块
    // 多行
    // 连续注释
    return x + y;
}

/* 测试块注释 */
fn main() -> i32 {
    /* 这是一个块注释 */
    let a: i32 = 5;
    let b: i32 = 3;
    /*
     * 多行块注释
     * 用于测试解析器是否正确跳过
     */
    return a - b;
}
